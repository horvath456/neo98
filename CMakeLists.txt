cmake_minimum_required(VERSION 3.16)
project(neo98_native_linux LANGUAGES CXX ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "This build requires Linux x86_64. The macOS build uses neo98.py.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(neo98_native_cpu OBJECT src/x86_cpu.cpp src/x86_address_space.cpp tools/host_native_cpu.cpp tools/direct_bridge.S)
target_compile_definitions(neo98_native_cpu PRIVATE NEO98_NO_UNICORN)
target_compile_options(neo98_native_cpu PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(test_opengl_arrays tools/test_opengl_arrays.cpp src/opengl_context.cpp)
target_compile_options(test_opengl_arrays PRIVATE -UNDEBUG)
add_test(NAME opengl_arrays COMMAND test_opengl_arrays)
add_executable(test_linux_scroll tools/test_linux_scroll.cpp)
target_compile_options(test_linux_scroll PRIVATE -UNDEBUG)
add_test(NAME linux_scroll COMMAND test_linux_scroll)
foreach(fixture IN ITEMS native_cpu cpu_self_patch cpu_nested_run)
    add_executable(test_${fixture} tools/test_${fixture}.cpp $<TARGET_OBJECTS:neo98_native_cpu>)
    target_compile_options(test_${fixture} PRIVATE -UNDEBUG)
    target_link_libraries(test_${fixture} PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
    add_test(NAME ${fixture} COMMAND test_${fixture})
    set_tests_properties(${fixture} PROPERTIES TIMEOUT 30 ENVIRONMENT "NEO98_CPU=native;NEO98_NATIVE_MODE=ptrace")
    add_test(NAME direct_${fixture} COMMAND test_${fixture})
    set_tests_properties(direct_${fixture} PROPERTIES TIMEOUT 30 ENVIRONMENT "NEO98_CPU=native;NEO98_NATIVE_MODE=direct")
endforeach()
add_executable(test_fiber tools/test_fiber.cpp src/fiber.cpp src/fiber_x86_64.S)
target_compile_options(test_fiber PRIVATE -UNDEBUG)
add_test(NAME fiber COMMAND test_fiber)
set_tests_properties(fiber PROPERTIES TIMEOUT 30)
add_executable(test_direct_bridge tools/test_direct_bridge.cpp tools/direct_bridge.S)
target_compile_options(test_direct_bridge PRIVATE -UNDEBUG)
add_test(NAME direct_bridge COMMAND test_direct_bridge)
set_tests_properties(direct_bridge PROPERTIES TIMEOUT 30)
add_executable(test_direct_yield tools/test_direct_yield.cpp src/fiber.cpp src/fiber_x86_64.S $<TARGET_OBJECTS:neo98_native_cpu>)
target_compile_options(test_direct_yield PRIVATE -UNDEBUG)
target_link_libraries(test_direct_yield PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
add_test(NAME direct_yield COMMAND test_direct_yield)
set_tests_properties(direct_yield PROPERTIES TIMEOUT 30 ENVIRONMENT "NEO98_CPU=native;NEO98_NATIVE_MODE=direct")

option(NEO98_BUILD_VULKAN "Build the Vulkan renderer and shader tests" OFF)
if(NEO98_BUILD_VULKAN)
    find_package(Vulkan REQUIRED)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    find_program(GLSLANG_VALIDATOR glslangValidator HINTS "${CMAKE_SOURCE_DIR}/build/toolchain/usr/bin" REQUIRED)
    set(SHADERS "${CMAKE_CURRENT_BINARY_DIR}/generated/vulkan_shaders.h")
    add_custom_command(OUTPUT "${SHADERS}"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/build_vulkan_shaders.py" --compiler "${GLSLANG_VALIDATOR}" --output "${SHADERS}"
        DEPENDS tools/build_vulkan_shaders.py tools/shaders/linux_basic.vert tools/shaders/linux_basic.frag
                tools/shaders/linux_clip.vert
                tools/shaders/linux_present.vert tools/shaders/linux_present.frag
                tools/shaders/linux_glide.vert tools/shaders/linux_glide.frag
                tools/shaders/linux_program.vert.glsl tools/shaders/linux_program_helpers.glsl
                tools/shaders/linux_program_vert_main.glsl tools/shaders/linux_program_frag_main.glsl)
    add_library(neo98_vulkan_program OBJECT tools/vulkan_program.cpp src/d3d_shader_msl.cpp "${SHADERS}")
    target_compile_definitions(neo98_vulkan_program PRIVATE NEO98_VULKAN_SHADERS_HEADER="${SHADERS}")
    add_executable(test_vulkan tools/test_vulkan.cpp tools/host_vulkan.cpp $<TARGET_OBJECTS:neo98_vulkan_program> "${SHADERS}")
    target_compile_definitions(test_vulkan PRIVATE NEO98_VULKAN_SHADERS_HEADER="${SHADERS}")
    target_compile_options(test_vulkan PRIVATE -UNDEBUG -Wall -Wextra -Wno-missing-field-initializers)
    target_link_libraries(test_vulkan PRIVATE Vulkan::Vulkan ${CMAKE_DL_LIBS})
    find_package(X11 REQUIRED)
    add_executable(test_vulkan_present tools/test_vulkan_present.cpp tools/host_vulkan.cpp $<TARGET_OBJECTS:neo98_vulkan_program> "${SHADERS}")
    target_compile_definitions(test_vulkan_present PRIVATE NEO98_VULKAN_SHADERS_HEADER="${SHADERS}")
    target_compile_options(test_vulkan_present PRIVATE -UNDEBUG)
    target_link_libraries(test_vulkan_present PRIVATE Vulkan::Vulkan X11::X11 ${CMAKE_DL_LIBS})
    add_test(NAME vulkan_present COMMAND test_vulkan_present)
    set_tests_properties(vulkan_present PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
    add_executable(test_vulkan_shader tools/test_d3d_shader_msl.cpp $<TARGET_OBJECTS:neo98_vulkan_program>)
    target_compile_definitions(test_vulkan_shader PRIVATE NEO98_SHADER_VULKAN)
    target_link_libraries(test_vulkan_shader PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME vulkan_shader COMMAND test_vulkan_shader)
    set_tests_properties(vulkan_shader PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
    add_test(NAME vulkan COMMAND test_vulkan)
    set_tests_properties(vulkan PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
endif()
